## API công khai mới (OSWidgetKit)

```swift
public enum URLPolicy {
    public static func isOpenable(_ url: URL) -> Bool        // http/https + host
    public static func isLoadable(_ url: URL) -> Bool        // https + host
    public static func openableURL(_ string: String?) -> URL?
    public static func loadableURL(_ string: String?) -> URL?
}
public enum Markdown { public static func attributed(_ source: String) -> AttributedString }
public enum CardMetrics { public static let width: CGFloat; padding: CGFloat; imageMaxHeight: CGFloat }
public struct CardView: View {
    public init(item: Item)
    public enum Variant { case notification, generic, fallback }
    public nonisolated static func variant(for content: ItemContent) -> Variant
}
public struct NotificationCardView: View { public init(item: Item, text: String?) }
public struct GenericCardView: View {
    public init(item: Item, blocks: [Block])
    public nonisolated static func renderedBlocks(_ blocks: [Block]) -> [Block]
}
public struct FallbackCardView: View { public init(item: Item) }
```

Trong app (không công khai): `Notification.Name.openItemListRequested` = `"local.oswidget.open-item-list"`. Khi nhận thông báo này, `MenuBarController` mở popover trên main queue và không làm gì nếu popover đang mở. Phase 5 sẽ post thông báo này.
