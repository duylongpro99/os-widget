// App/AppEnvironment.swift
@MainActor final class AppEnvironment {
    let config: AppConfig
    let tokenStore: KeychainTokenStore
    let store: ItemStore
    init(config: AppConfig, tokenStore: KeychainTokenStore, store: ItemStore)
    static func live() -> AppEnvironment
}

// App/SyncWiring.swift
@MainActor enum SyncWiring {
    static func start(env: AppEnvironment)
}

// App/MenuBar/MenuBarController.swift

    init(env: AppEnvironment)
    private(set) var isListOpen: Bool
    func anchorRect() -> NSRect?

    init(env: AppEnvironment, anchor: @escaping () -> NSRect?, isListOpen: @escaping () -> Bool)
